f_num(float x) => na(x) ? "null" : str.tostring(x, format.mintick)
f_num2(float x) => na(x) ? "null" : str.tostring(x, "#.##")

// Previous CONFIRMED bar of whichever timeframe request.security() is evaluating.
// Used for all higher-timeframe snapshots to avoid feeding a still-forming HTF bar.
f_prev_snapshot() =>
    float e20 = ta.ema(close, 20)
    float e50 = ta.ema(close, 50)
    float e200 = ta.ema(close, 200)
    float rv = ta.rsi(close, 14)
    float av = ta.atr(14)
    float vw = ta.vwap(hlc3)
    float sh = ta.highest(high[1], 20)
    float sl = ta.lowest(low[1], 20)
    float vm = ta.sma(volume, 20)
    [open[1], high[1], low[1], close[1], e20[1], e50[1], e200[1], rv[1], av[1], vw[1], sh[1], sl[1], close[4], open[2], close[2], volume[1], vm[1]]

// Current 1-minute snapshot. The alert itself only fires after the chart's 1m bar closes.
f_current_snapshot() =>
    float e20 = ta.ema(close, 20)
    float e50 = ta.ema(close, 50)
    float e200 = ta.ema(close, 200)
    float rv = ta.rsi(close, 14)
    float av = ta.atr(14)
    float vw = ta.vwap(hlc3)
    float sh = ta.highest(high[1], 20)
    float sl = ta.lowest(low[1], 20)
    float vm = ta.sma(volume, 20)
    [open, high, low, close, e20, e50, e200, rv, av, vw, sh, sl, close[3], open[1], close[1], volume, vm]

    '{"tf":"' + tf + '","o":' + f_num(o) + ',"h":' + f_num(h) + ',"l":' + f_num(l) + ',"c":' + f_num(c) + ',"ema20":' + f_num(e20) + ',"ema50":' + f_num(e50) + ',"ema200":' + f_num(e200) + ',"rsi":' + f_num2(rv) + ',"atr":' + f_num(av) + ',"vwap":' + f_num(vw) + ',"swing_hi":' + f_num(sh) + ',"swing_lo":' + f_num(sl) + ',"c3":' + f_num(c3) + ',"prev_o":' + f_num(po) + ',"prev_c":' + f_num(pc) + ',"vol":' + f_num(vol) + ',"vma20":' + f_num(vma20) + '}'
